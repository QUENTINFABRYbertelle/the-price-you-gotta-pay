  <script>

    const MAX_PLAYERS = 10;

    const players = [];

    const bids = []; // { player, type, value, numericValueForSort }


    const playerNameInput = document.getElementById('playerName');

    const addPlayerBtn = document.getElementById('addPlayerBtn');

    const signupMsg = document.getElementById('signupMsg');

    const playerList = document.getElementById('playerList');

    const bidPlayerSelect = document.getElementById('bidPlayer');


    const bidTypeSelect = document.getElementById('bidType');

    const moneyBidBlock = document.getElementById('moneyBidBlock');

    const crazyBidBlock = document.getElementById('crazyBidBlock');

    const moneyBidInput = document.getElementById('moneyBid');

    const crazyBidInput = document.getElementById('crazyBid');

    const placeBidBtn = document.getElementById('placeBidBtn');

    const bidMsg = document.getElementById('bidMsg');


    const currentWinnerEl = document.getElementById('currentWinner');

    const currentBidEl = document.getElementById('currentBid');

    const leaderboardBody = document.getElementById('leaderboardBody');


    function refreshPlayerUI() {

      // player list

      playerList.innerHTML = '';

      players.forEach(p => {

        const li = document.createElement('li');

        li.textContent = p;

        playerList.appendChild(li);

      });


      // select

      bidPlayerSelect.innerHTML = '';

      players.forEach(p => {

        const opt = document.createElement('option');

        opt.value = p;

        opt.textContent = p;

        bidPlayerSelect.appendChild(opt);

      });

    }


    function refreshLeaderboard() {

      leaderboardBody.innerHTML = '';


      // sort bids: money first by numeric desc, then crazy bids after

      const sorted = [...bids].sort((a, b) => {

        // money vs crazy: money wins

        if (a.type === 'money' && b.type === 'crazy') return -1;

        if (a.type === 'crazy' && b.type === 'money') return 1;

        // both money

        if (a.type === 'money' && b.type === 'money') {

          return b.numericValueForSort - a.numericValueForSort;

        }

        // both crazy: keep insertion order

        return 0;

      });


      sorted.forEach((bid, index) => {

        const tr = document.createElement('tr');

        const rankTd = document.createElement('td');

        const playerTd = document.createElement('td');

        const bidTd = document.createElement('td');

        const typeTd = document.createElement('td');


        rankTd.textContent = index + 1;

        playerTd.textContent = bid.player;

        bidTd.textContent = bid.value;

        typeTd.textContent = bid.type === 'money' ? 'Money' : 'Outrageous';


        // highlight top bid

        if (index === 0) {

          playerTd.classList.add('highlight');

          bidTd.classList.add('highlight');

        }


        tr.appendChild(rankTd);

        tr.appendChild(playerTd);

        tr.appendChild(bidTd);

        tr.appendChild(typeTd);

        leaderboardBody.appendChild(tr);

      });


      if (sorted.length === 0) {

        currentWinnerEl.textContent = 'Current winner: none';

        currentBidEl.textContent = 'Current bid: none';

      } else {

        const top = sorted[0];

        currentWinnerEl.textContent = 'Current winner: ' + top.player;

        currentBidEl.textContent = 'Current bid: ' + top.value + ' (' + (top.type === 'money' ? '$' : 'outrageous') + ')';

      }

    }


    addPlayerBtn.addEventListener('click', () => {

      const name = playerNameInput.value.trim();

      if (!name) {

        signupMsg.textContent = 'Enter a name.';

        return;

      }

      if (players.length >= MAX_PLAYERS) {

        signupMsg.textContent = 'Maximum of ' + MAX_PLAYERS + ' players reached.';

        return;

      }

      if (players.includes(name)) {

        signupMsg.textContent = 'That name is already taken.';

        return;

      }

      players.push(name);

      playerNameInput.value = '';

      signupMsg.textContent = 'Player added.';

      refreshPlayerUI();

    });


    bidTypeSelect.addEventListener('change', () => {

      const type = bidTypeSelect.value;

      if (type === 'money') {

        moneyBidBlock.style.display = 'block';

        crazyBidBlock.style.display = 'none';

      } else {

        moneyBidBlock.style.display = 'none';

        crazyBidBlock.style.display = 'block';

      }

    });


    placeBidBtn.addEventListener('click', () => {

      bidMsg.textContent = '';


      if (players.length === 0) {

        bidMsg.textContent = 'No players yet. Sign up first.';

        return;

      }


      const player = bidPlayerSelect.value;

      const type = bidTypeSelect.value;


      if (!player) {

        bidMsg.textContent = 'Choose a player.';

        return;

      }


      if (type === 'money') {

        const val = Number(moneyBidInput.value);

        if (isNaN(val) || val < 500) {

          bidMsg.textContent = 'Money bid must be at least $500.';

          return;

        }

        bids.push({

          player,

          type: 'money',

          value: '$' + val.toLocaleString(),

          numericValueForSort: val

        });

        bidMsg.textContent = 'Money bid placed.';

        moneyBidInput.value = '';

      } else {

        const text = crazyBidInput.value.trim();

        if (!text) {

          bidMsg.textContent = 'Describe your outrageous bid.';

          return;

        }


        // Give crazy bids a pseudo-random "value" so they can sometimes beat money

        const crazyScore = 999999 + Math.floor(Math.random() * 1000000);


        bids.push({

          player,

          type: 'crazy',

          value: text,

          numericValueForSort: crazyScore

        });

        bidMsg.textContent = 'Outrageous bid placed.';

        crazyBidInput.value = '';

      }


      refreshLeaderboard();

    });


    // Initial UI

    refreshPlayerUI();

    refreshLeaderboard();

  </script>
